from src.data import load_minute_data, resample_to_15m
from src.halftrend import calculate_halftrend
from src.trading import generate_long_only_trades


DATA_PATH = (
    r"C:\Users\beqmd\Documents\QuantResearch"
    r"\data\NIFTY_50_minute.csv"
)


df = resample_to_15m(
    load_minute_data(DATA_PATH)
)

ht = calculate_halftrend(df)

positions, trade_results = generate_long_only_trades(ht)


events = []

in_position = False
entry_time = None
entry_price = None


for i in range(1, len(ht)):

    # Signals occur at bar close.
    # Execution occurs at the following bar open.

    if (
        bool(ht["buy_signal"].iloc[i - 1])
        and not in_position
    ):

        entry_time = ht.index[i]
        entry_price = float(ht["open"].iloc[i])

        in_position = True

    elif (
        bool(ht["sell_signal"].iloc[i - 1])
        and in_position
    ):

        exit_time = ht.index[i]
        exit_price = float(ht["open"].iloc[i])

        pnl = exit_price - entry_price

        events.append(
            {
                "trade": len(events) + 1,
                "entry_time": entry_time,
                "entry_price": entry_price,
                "exit_time": exit_time,
                "exit_price": exit_price,
                "pnl": pnl,
            }
        )

        in_position = False
        entry_time = None
        entry_price = None


import pandas as pd


ledger = pd.DataFrame(events)

output = "results/halftrend_full_trade_ledger.csv"

ledger.to_csv(
    output,
    index=False,
)

print("=" * 60)
print("HALFTREND — FULL TRADE LEDGER")
print("=" * 60)
print()

print(f"Completed trades: {len(ledger):,}")
print(f"Saved: {output}")
